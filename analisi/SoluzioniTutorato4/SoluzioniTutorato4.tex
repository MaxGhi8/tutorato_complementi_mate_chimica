\begin{exe}
    Verificate che la funzione $f(x,y) = \alpha x + \beta y + \gamma$ è armonica.
\end{exe}
\begin{sol}
    $f_{xx} = 0, f_{yy} = 0 \implies \Delta f = 0$.
\end{sol}

\begin{exe}
    Verificate funzioni armoniche...
\end{exe}
\begin{sol}
    $\Delta(e^{kx}\cos(ky)) = k^2 e^{kx}\cos(ky) - k^2 e^{kx}\cos(ky) = 0$.
\end{sol}

\begin{exe}
    Laplaciano in coordinate polari...
\end{exe}
\begin{sol}
    (1) $\frac{\theta^2 + 2}{\rho}$; (2) $2 + 2\theta/\rho$.
\end{sol}

\begin{exe}
    Determina se le forme differenziali sono esatte...
\end{exe}
\begin{sol}
    (1) è esatta, potenziale $U = xyzw$;
    (2) è esatta, potenziale $U = \sin(a)b \cos(c) + ac^2$;
    (3) è esatta, potenziale $U = \sin(u) - \cos(v)$;
    (4) è esatta, potenziale $U = \sin(uv)$.
\end{sol}

\begin{exe}
    Campo conservativo $F(x,y,z) = (3x^2 y^2 z, 2x^3 y z^k, x^3 y^2)$.
\end{exe}
\begin{sol}
    $k=1$. Potenziale $U = x^3 y^2 z$.
\end{sol}
