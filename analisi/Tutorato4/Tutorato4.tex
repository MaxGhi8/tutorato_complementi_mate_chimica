\section*{Integrali multipli}

\begin{exe}
    Calcola i seguenti integrali multipli.
    \begin{enumerate}
        \item $I_1 = \iint_{[1,2]\times[-2,-1]} \left(\frac{y^2}{x}\right) dx \, dy$;
        \item $I_2 = \iiint_{[0,1]^3} (y e^{xy} - z) dx \, dy \, dz$;
        \item $I_3 = \int_{[-1,1]^4} (1 - xyzw) dx \, dy \, dz \, dw$.
    \end{enumerate}
\end{exe}

\begin{exe}
    Definisci un insieme $y$-semplice che non sia un rettangolo, definisci un insieme $x$-semplice che non sia un rettangolo, poi definisci un insieme che non sia né $y$-semplice né $x$-semplice.
\end{exe}

\begin{exe}
    Consideriamo gli insiemi $A = \{(x,y) \in \mathbb{R}^2 : 0 \leq x \leq y^2 \leq 1\}$ e $B = \{(x,y) \in \mathbb{R}^2 : x \in [3,4]; x \leq y < x^2-2\}$. Sapendo che la funzione $f : \mathbb{R}^2 \to \mathbb{R}$ verifica $\iint_A f(x,y) \, dx dy = 1$ e $\iint_B f(x,y) \, dx dy = -1$, calcola $\iint_{A \cup B} 5f(x,y) \, dx dy$.
\end{exe}

\begin{exe}
    Calcola
    \begin{enumerate}
        \item il volume del solido avente per base l'insieme $\{(x,y,1) : 1 \leq x^2 + y^2 \leq 4\}$ e la cui altezza $z$ soddisfa la disequazione $z \leq \left| \frac{xy}{x^2+y^2} \right| + 1$;
        \item il volume del solido $S = \{(x,y,z) : 0 \leq z \leq 1; x^2 + \frac{y^2}{2} = z^2 \}$.
    \end{enumerate}
\end{exe}

\begin{exe}
    Calcola il seguente integrale doppio:
    \[ \iint_\Omega xy^3 \, dx dy \]
    dove $\Omega = \left\{(x,y) \in \mathbb{R}^2 : \frac{x^2}{4} + \frac{y^2}{9} \leq 1\right\}$.
\end{exe}

\begin{exe}
    Calcola il seguente integrale triplo:
    \[ \iiint_\Omega x^2(2-z) \, dx dy dz \]
    dove $\Omega = \{(x,y,z) \in \mathbb{R}^3 : 0 \leq z \leq 2, x^2+y^2 \leq z\}$.
\end{exe}

\begin{exe}
    Dato l'insieme
    \[ \Omega = \{(x, y) : x \in [0, 2\pi), \  \sin(x) \le y \le cos(x) + 2\},
    \]
    calcolare $\int \int _{\Omega} 2y \, dx dy$.
\end{exe}

\begin{exe}
    Dato l'insieme
    \[ \Omega = \{(x, y) : y \in [-1, 1], \  y-1 \le x \le y^2\},
    \]
    calcolare $\int \int _{\Omega} y \, dx dy$.
\end{exe}
\begin{exe}
    Dato $T$ il triangolo di vertici $(-1, 0)$, $(0, 2)$ e $(2,0)$ calcolare $\int \int_{T} \frac{y}{1+x}\, dx dy$
\end{exe}
\begin{exe}
    Dato $T$ il triangolo di vertici $(0, 0)$, $(1, \sqrt{3})$ e $(1,0)$ calcolare $\int \int_{T} xy \sqrt{x^2 + y^2}\, dx dy$
\end{exe}
