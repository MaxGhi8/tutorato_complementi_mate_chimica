\begin{sol}
    \begin{enumerate}
        \item $I_1 = \iint_{[1,2]\times[-2,-1]} \left(\frac{y^2}{x}\right) dx \, dy = \int_1^2 \frac{1}{x} dx \int_{-2}^{-1} y^2 dy = [\ln x]_1^2 [\frac{y^3}{3}]_{-2}^{-1} = \ln(2) \frac{7}{3}$.
        \item $I_2 = \iiint_{[0,1]^3} (y e^{xy} - z) dx \, dy \, dz = \int_0^1 \left( \int_0^1 (e^{xy} - zx)|_0^1 dy \right) dz = \int_0^1 \int_0^1 (e^y - z - 1) dy dz = \int_0^1 (e - 1 - z - 1) dz = e - 2 - 1/2 = e - 5/2$.

\begin{sol}

    \[ I = \int_0^2 (2-z) \frac{\pi z^2}{4} dz = \frac{\pi}{4} \left[\frac{2z^3}{3} - \frac{z^4}{4}\right]_0^2 = \frac{\pi}{4} (\frac{16}{3} - 4) = \frac{\pi}{3}. \]
